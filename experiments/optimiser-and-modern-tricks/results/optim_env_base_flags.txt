waitmem() {  # wait until GPU N has at least $1 MiB free
  local need=${1:-40000}
  while true; do
    f=$(nvidia-smi --query-gpu=memory.free --format=csv,noheader,nounits -i N)
    [ "$f" -ge "$need" ] && break
    sleep 30
  done
}
