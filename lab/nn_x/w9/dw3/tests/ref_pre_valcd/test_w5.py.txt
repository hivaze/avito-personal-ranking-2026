def load_parse(path):
    tree = ast.parse(open(path).read())
    fn = [n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == "parse"][0]
    ns = {"argparse": argparse, "POOL_P_DEFAULT": CA.POOL_P_DEFAULT}
    exec(compile(ast.Module(body=[fn], type_ignores=[]), "train_parse", "exec"), ns)

    def parse(argv):
        sys.argv = ["train.py"] + argv
        return ns["parse"]()
    return parse

class StripWave5(ast.NodeTransformer):
    def __init__(self):
        self.ifs, self.ifexps, self.other = collections.Counter(), [], []

    def visit_ImportFrom(self, n):
        if n.module == "cd_arms5" or (n.module == "common" and [a.name for a in n.names] == ["ndcg"]):
            self.other.append(f"from {n.module} import {[a.name for a in n.names]}")
            return None
        return n

    def visit_Expr(self, n):
        v = n.value
        if (isinstance(v, ast.Call) and isinstance(v.func, ast.Attribute) and v.func.attr == "add_argument"
                and v.args and isinstance(v.args[0], ast.Constant) and v.args[0].value in W5_FLAGS):
            self.other.append(f"add_argument {v.args[0].value}")
            return None
        return self.generic_visit(n)

    def visit_Assign(self, n):
        u = ast.unparse(n)
        if u == "trunc, cliplog = (None, None)":
            self.other.append(u)
            return None
        if u == "gn = torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)":
            self.other.append("gn = clip_grad_norm_ -> bare call")
            return ast.copy_location(ast.Expr(value=n.value), n)
        return self.generic_visit(n)

    def visit_If(self, n):
        t = ast.unparse(n.test)
        if t in G5_IF:
            self.ifs[t] += 1
            return None
        return self.generic_visit(n)

    def visit_IfExp(self, n):
        t = ast.unparse(n.test)
        if t in G5_IFEXP:
            self.ifexps.append((t, ast.unparse(n.orelse)))
            return self.visit(n.orelse)
        return self.generic_visit(n)


def strip_doc(tree):
    if tree.body and isinstance(tree.body[0], ast.Expr) and isinstance(tree.body[0].value, ast.Constant):
        tree.body = tree.body[1:]
    return tree


def dry_cmds(script, gpu, jobs, rx):
    out = subprocess.run(["bash", script, str(gpu), jobs], capture_output=True, text=True, env={**os.environ, "DRY": "1"})
    cmds = {}
    for line in out.stdout.splitlines():
        if " train.py " in line:
            m = re.search(rx, line)
            cmds[m.group(0)] = line.split(" train.py ", 1)[1].split()
    return out.returncode, cmds


def diff_ns(a, b, skip=("variant",)):
    return {k: (a[k], b[k]) for k in a if k not in skip and a[k] != b[k]}

    check("feeds.parquet cand_off == exclusive cumsum of M (the artifact builder used cumsum offsets)",
          np.array_equal(off, np.concatenate([[0], np.cumsum(M)[:-1]])))
    pos = np.load(f"{ROOT}/data/flat/train/cand_pos.npy").astype(np.int64)
    rk = np.load(C5.RKEND_PATH)
    rank = np.load(f"{C5.ART}/train_rank_true.npy")
    N = int(M.sum())
    check("rkend / rank arrays: int16, one value per train candidate in flat order",
          rk.shape == (N,) and rank.shape == (N,) and pos.shape == (N,) and rk.dtype == np.int16, (rk.shape, N))
    fid = np.repeat(np.arange(n, dtype=np.int64), M)
    assert pos.min() >= 0 and pos.max() < 32768
    key = fid * 32768 + (32767 - pos)  # within a feed: descending shown position
    # tie rule of 15_nn_artifacts.py (lexsort by (feed, pos), stable): among equal shown positions the LATER flat row
    # is closer to the end, i.e. rkend_i = #{pos_j > pos_i} + #{pos_j == pos_i, j > i}. A stable sort of the REVERSED
    # arrays reproduces exactly that order.
    o = N - 1 - np.argsort(key[::-1], kind="stable")
    ks = key[o]
    dup = np.flatnonzero(ks[1:] == ks[:-1])
    tied = np.unique(np.concatenate([o[dup], o[dup + 1]])) if len(dup) else np.zeros(0, np.int64)
    P(f"  info: duplicated shown positions: {len(dup)} pairs in {len(np.unique(fid[tied]))} feeds ({len(tied)} rows, "
      f"{int((rk[tied] <= 2).sum())} of them at rkend <= 2): ranked by the flat-order tie rule above")
    check("duplicated shown positions are negligible (< 50 feeds, < 1000 rows)",
          len(np.unique(fid[tied])) < 50 and len(tied) < 1000)
    del ks
    re_ = np.empty(N, np.int64)
    re_[o] = np.arange(N) - off[fid[o]]
    del o, key
    check("train_rkend_true.npy == independent recomputation (#later-shown candidates of the same feed; flat-order "
          "ties), all 246.7M candidates", np.array_equal(re_, rk.astype(np.int64)),
          f"mismatches {int((re_ != rk).sum())}; {time.time() - t0:.0f}s")
    check("rank + rkend == M - 1 for every candidate (train_rank_true.npy consistent)",
          np.array_equal(rank.astype(np.int64) + rk.astype(np.int64), M[fid] - 1))
    nz = np.bincount(fid[rk == 0], minlength=n)
    check("exactly one rkend-0 candidate per feed", (nz == 1).all())
    P(f"  info: rkend != M-1-raw pos in {float(np.mean(rk.astype(np.int64) != np.maximum(M[fid] - 1 - pos, 0))):.3f} of "
      f"candidates; feeds with position gaps {float(np.mean(np.bincount(fid, pos, n) != M * (M - 1) / 2)):.3f}")
    del re_, rank, pos
    y = np.load(f"{ROOT}/data/flat/train/cand_target.npy").astype(np.float64)
    g = C5.trunc_groups(M, meta["p"])
    gh = g.copy()
    gh[meta["hold"]] = 3
    gc = gh[fid]
    j = json.load(open(C5.LIFT_PATH))
    rec = {}
    for k, nm in enumerate(C5.GROUPS):
        m = gc == k
        base = y[m & (rk >= 3)].mean()
        rec[nm] = [float(np.log(y[m & (rk == r)].mean() / base)) for r in range(3)]
    d = max(abs(rec[nm][r] - j[nm][f"rkend{r}"]) for nm in C5.GROUPS for r in range(3))
    check("trunc_lift.json production lifts == recomputed from train labels with the NN-rule groups (|d| < 1e-9)",
          d < 1e-9, {nm: np.round(v, 4).tolist() for nm, v in rec.items()})
    want = {"prod_M<50": [1.00, 0.54, 2.01], "prod_M>=50": [2.81, -0.20, -0.22]}
    got = {nm: [round(j[nm][f"rkend{r}"], 2) for r in range(3)] for nm in C5.GROUPS}
    check("lift table == the analysis numbers (prod M<50 1.00/0.54/2.01, prod M>=50 2.81/-0.20/-0.22)", got == want, got)
    L = C5.load_lifts()
    check("load_lifts(): rows prod M<50 / prod M>=50 from the json, random-like row and rkend>=3 column zero",
          np.allclose(L[0, :3], rec["prod_M<50"]) and np.allclose(L[1, :3], rec["prod_M>=50"]) and not L[2].any()
          and not L[:, 3].any())
    RES["lifts"] = rec
    del y, gc, fid, rk


def c2_trunc(meta):
    P("\n== C2 TRUNC table on the real feeds + TruncOffset.offset on MiniSplit batches")
    L = C5.load_lifts()
    M, p, ps, hold = meta["M"], meta["p"], meta["p_soft"], meta["hold"]
    coef, g = C5.trunc_coef(1.0, M, p.astype(np.float64), ps, L, hold)
    pool = (p > 0.5) & (M >= 50)
    check("pool-eligible (NN p > 0.5) and holdout feeds: offset 0; column rkend>=3: 0",
          not coef[pool].any() and not coef[hold].any() and not coef[:, 3].any(),
          f"{int(pool.sum())} pool-eligible incl. {len(hold)} holdout")
    w = 1.0 - ps
    m0, m1 = (M < 50), (M >= 50) & ~pool
    ok0 = np.allclose(coef[m0, :3], (L[0, :3][None, :] * w[m0, None]).astype(np.float32), rtol=0, atol=0)
    ok1 = np.allclose(coef[m1, :3], (L[1, :3][None, :] * w[m1, None]).astype(np.float32), rtol=0, atol=0)
    check("M<50 rows == L[prod M<50] * (1 - p_soft) and M>=50 production rows == L[prod M>=50] * (1 - p_soft) (exact)",
          ok0 and ok1 and (g[m0] == 0).all() and (g[m1] == 1).all())
    hid = m0 & (ps > 0.5)
    RES["trunc_weights"] = dict(M_lt50_mean_w=float(w[m0].mean()), M_lt50_hidden_expl=int(hid.sum()),
                                M_lt50_hidden_expl_mean_w=float(w[hid].mean()), prod_M_ge50_mean_w=float(w[m1].mean()))
    P(f"  info: mean (1 - p_soft): M<50 {w[m0].mean():.3f} (of which p_soft > 0.5: {int(hid.sum())} feeds, mean weight "
      f"{w[hid].mean():.3f}); production M>=50 {w[m1].mean():.3f}; random-like 0")
    feeds = pick_train_feeds(n_each=40, seed=5)
    ms = MiniSplit("train", feeds, xt_names=None)
    rk_full = np.load(C5.RKEND_PATH, mmap_mode="r")
    crow = np.concatenate([np.arange(meta["off"][f], meta["off"][f] + M[f]) for f in ms.src_feeds])
    to = object.__new__(C5.TruncOffset)
    to.rk3 = torch.from_numpy(np.minimum(np.asarray(rk_full[crow]), 3).astype(np.uint8))
    to.coef = torch.from_numpy(coef[ms.src_feeds])
    bad, nb, nz = 0, 0, 0
    holdset = set(hold.tolist())
    for fb in ms.batches(np.arange(ms.n), 8192, 64, shuffle=True, rng=np.random.default_rng(0)):
        b = ms.gather(fb, with_pos=True)
        o_ = to.offset(b).numpy()
        ci, cm = b["ci"].numpy(), b["cmask"].numpy()
        for i, f in enumerate(fb):
            src = ms.src_feeds[f]
            for jj in range(o_.shape[1]):
                if not cm[i, jj]:
                    exp = 0.0
                else:
                    r = int(rk_full[crow[ci[i, jj]]])
                    gg = 0 if M[src] < 50 else (2 if p[src] > 0.5 else 1)
                    exp = 0.0 if (r > 2 or gg == 2 or src in holdset) else float(np.float32((1.0 * L[gg, r]) * w[src]))
                bad += int(o_[i, jj] != exp)
                nz += int(exp != 0)
        nb += 1
    check(f"TruncOffset.offset == brute-force per-candidate loop on {nb} MiniSplit batches (0 on padding / random-like)",
          bad == 0 and nz > 0, f"mismatches {bad}, non-zero offsets {nz}")


def c3_auxf(meta):
    P("\n== C3 AUXF constant from the real batch construction")
    tr = FakeTr(meta["M"])
    p = meta["p"].astype(np.float64)
    cs, mults = [], []
    for seed in (0, 23, 101):
        bs = tr.batches(meta["feeds"], 32768, 512, shuffle=True, rng=np.random.default_rng(seed))
        mult, c = C5.auxf_multiplier(bs, p, C5.AUXF_C, 512)
        cs.append(c)
        mults.append(mult)
    check("exact c (batches per epoch / sum_b sum_f (1-p_f) / 512) == 1.695 +- 0.002 for 3 shuffles; realised "
          "multiplier with c = 1.695 within 1 +- 0.002", all(abs(c - 1.695) < 0.002 for c in cs)
          and all(abs(m - 1) < 0.002 for m in mults), f"c {np.round(cs, 4).tolist()}, multiplier {np.round(mults, 4).tolist()}")
    pool = meta["feeds"][p[meta["feeds"]] > 0.5]
    pb = tr.batches(pool, 32768, 512, shuffle=True, rng=np.random.default_rng(1))
    pm, _ = C5.auxf_multiplier(pb, p, C5.AUXF_C, 512)
    bs = tr.batches(meta["feeds"], 32768, 512, shuffle=True, rng=np.random.default_rng(0))
    rows = collections.defaultdict(list)
    for fb in bs:
        Mp = int(-(-meta["M"][fb].max() // 16) * 16)
        rows[Mp].append((len(fb), (1 - p[fb]).sum()))
    prof = {mp: (round(0.5 / (np.mean([b for b, _ in v]) / 512), 2),
                 round(0.5 * C5.AUXF_C * np.mean([s for _, s in v]) / 512 / (np.mean([b for b, _ in v]) / 512), 2))
            for mp, v in sorted(rows.items())[::3]}
    P(f"  info: m02 pool batches under AUXF get an aux multiplier of {pm:.3f} (live recipe 1.0); aux:main per batch by "
      f"Mpad (live, AUXF) {prof}")
    RES["auxf"] = dict(c=cs, mult=mults, pool_mult=pm)


# ================================================================================================= GPU part
def clone(m):
    for mod in m.modules():
        for k in ("aux_out", "intent_logp"):
            mod.__dict__.pop(k, None)
    return copy.deepcopy(m)


def aux_stmt(path):
    tree = ast.parse(open(path).read())
    st = [n for n in ast.walk(tree) if isinstance(n, ast.Assign) and ast.unparse(n).startswith("loss = loss + aux_w(step)")
          and "tgt * lp" in ast.unparse(n)]
    assert len(st) == 1
    return compile(ast.Module(body=[st[0]], type_ignores=[]), "aux_stmt", "exec"), ast.unparse(st[0])


def g1_auxf():
    import model as M5
    P("\n== G1 AUXF: train.py's aux statement exec'd on real train-mode batches (F7 @77.5k)")
    ck = torch.load(INIT, map_location="cpu", weights_only=False)
    base = M5.Ranker(**ck["cfg"])
    base.load_state_dict(ck["state_dict"])
    base.to(DEV)
    feeds = pick_train_feeds(n_each=48, seed=3)
    ms = MiniSplit("train", feeds, xt_names=ck["cfg"]["xt_names"]).to(DEV)
    new_c, new_s = aux_stmt(TRAIN_PY)
    ref_c, ref_s = aux_stmt(REF_TRAIN)
    P(f"  statement: {new_s}")
    prn = ms.prand.cpu().numpy()
    for nm, ix in (("regular", np.where((prn < 0.1) & (ms.M_np >= 50))[0][:64]),
                   ("pool", np.where((prn > 0.5) & (ms.M_np >= 50))[0][:64]), ("M<50", np.where(ms.M_np < 50)[0][:64])):
        mdl = clone(base)
        mdl.train()
        torch.manual_seed(5)
        b = ms.gather(ix, with_pos=True)
        mdl(b, mdl.dense(b), use_pos=True)
        m = b["cmask"]
        tgt = torch.softmax((-b["pos"].float() / 10.0).masked_fill(~m, -1e4), 1) * m
        lp = torch.log_softmax(mdl.aux_out.masked_fill(~m, -1e4), 1)
        fw_aux = 1 - ms.prand[b["fidx"]]
        out = {}
        for norm in ("batch", "feed"):
            a = argparse.Namespace(aux_norm=norm, aux_norm_c=1.695, max_feeds=512)
            ns = dict(a=a, aux_w=lambda s: 0.5, step=0, loss=torch.zeros((), device=DEV), tgt=tgt, lp=lp, fw_aux=fw_aux,
                      torch=torch)
            exec(new_c, ns)
            out[norm] = ns["loss"]
        ns = dict(aux_w=lambda s: 0.5, step=0, loss=torch.zeros((), device=DEV), tgt=tgt, lp=lp, fw_aux=fw_aux, torch=torch)
        exec(ref_c, ns)
        ce = -(tgt * lp).sum(1)
        wm = 0.5 * (ce * fw_aux).sum() / fw_aux.sum().clamp_min(1e-6)
        ff = 0.5 * 1.695 * (ce * fw_aux).sum() / 512
        ratio = float(out["feed"] / out["batch"])
        check(f"{nm} batch (B {len(ix)}): batch norm == r6 statement (bitwise) == weighted mean; feed norm == 1.695 * "
              f"sum / 512", torch.equal(out["batch"], ns["loss"]) and torch.allclose(out["batch"], wm, rtol=1e-6)
              and torch.allclose(out["feed"], ff, rtol=1e-6),
              f"aux batch {out['batch'].item():.5f} feed {out['feed'].item():.5f} (ratio {ratio:.3f} = 1.695 * "
              f"sum(1-p) / 512 = {1.695 * fw_aux.sum().item() / 512:.3f})")

def outputs(v):
    fs = [f"{W5}/logs/{v}.log", f"{W5}/logs/{v}.json", f"{W5}/logs/{v}_cliplog.npz", f"{W5}/logs/{v}_holdout.stdout"]
    for x in (v, f"{v}@last"):
        fs += [f"{W5}/ckpt/{x}.pt", f"{W5}/preds/nn_{x}_trainonly_val.npy", f"{W5}/preds/nn_{x}_trainonly_test.npy",
               f"{W5}/preds/holdout/nn_{x}_holdout.npy"]
    return fs


def clean(v):
    for f in outputs(v):
        if os.path.exists(f):
            os.remove(f)


def e2e_run(train_file, v, extra, no_test=True):
    rec = f"{SCR}/rec_{v}.npz"
    args = E2E.format(init=INIT).split() + (["--no_test"] if no_test else []) + extra
    cmd = [PY, f"{HERE}/e2e_w5.py", train_file, rec, "--variant", v] + args
    t0 = time.time()
    r = subprocess.run(cmd, capture_output=True, text=True, cwd=W5, env={**os.environ, "OMP_NUM_THREADS": str(NT)})
    open(f"{SCR}/{v}.stdout", "w").write(r.stdout + "\n" + r.stderr)
    P(f"  ran {v} ({time.time() - t0:.0f}s, rc {r.returncode})")
    assert r.returncode == 0, r.stderr[-3000:]
    return dict(np.load(rec, allow_pickle=True)), open(f"{W5}/logs/{v}.log").read()


def metric_lines(log):
    return [re.sub(r" [\d.]+ min", "", ln) for ln in log.splitlines() if ln.startswith("step ") or ln.startswith("BEST")]


def batches_of(rec, key):
    sh, out, k = rec["shapes"], [], 0
    for b_, m_ in sh:
        out.append(rec[key][k:k + b_ * m_].reshape(b_, m_))
        k += b_ * m_
    return out


def ck_equal(a, b):
    ca = torch.load(a, map_location="cpu", weights_only=False)["state_dict"]
    cb = torch.load(b, map_location="cpu", weights_only=False)["state_dict"]
    return ca.keys() == cb.keys() and all(torch.equal(ca[k], cb[k]) for k in ca)

          f"gn median {np.median(cl['gn']):.3f} max {cl['gn'].max():.3f}; split {cl['split'][:, 3:].round(3).tolist()}")
    # arms ON
    rc_, lc = e2e_run(TRAIN_PY, names[3], ["--trunc_off", "1", "--aux_norm", "feed", "--cliplog", "10", "--save_last",
                                           "--save_holdout", "--select_min_step", "39"], no_test=False)
    check("arms ON: same training-batch sequence as flags OFF (arms share the data order)",
          np.array_equal(rc_["fidx"], rb["fidx"]) and int(rc_["n_loss"]) == int(rc_["n_model"]) == 39)

    M, p, ps = meta["M"], meta["p"], meta["p_soft"]
    lgm, lgl, cis = batches_of(rc_, "lg_model"), batches_of(rc_, "lg_loss"), batches_of(rc_, "ci")

        got = lgl[i]
        bad += int(not np.array_equal(got, lgm[i] + exp))
        d = got - lgm[i]
        rnd = np.array([(M[f] >= 50 and p[f] > 0.5) for f in fb])
        nz_prod += int(np.count_nonzero(d[~rnd]))
        nz_rand += int(np.count_nonzero(d[rnd])) if rnd.any() else 0
    check("arms ON: for EVERY training batch, logits into the main listwise loss == model logits + brute-force offset "
          "a * L[group, rkend] * (1 - p_soft) (exact, fp32)", bad == 0, f"{bad} mismatching batches")
    check("arms ON: offsets non-zero on production feeds, zero on every random-like (pool) feed",
          nz_prod > 0 and nz_rand == 0, f"non-zero production cells {nz_prod}, random-like {nz_rand} "
                                        f"({n_rand_feeds} random-like feed rows)")
    jj = json.load(open(f"{W5}/logs/{names[3]}.json"))
    check("arms ON: selection only at step >= 39 (best_step 39); @last checkpoint == selected checkpoint (bitwise)",
          jj["best_step"] == 39 and ck_equal(f"{W5}/ckpt/{names[3]}.pt", f"{W5}/ckpt/{names[3]}@last.pt"), jj["best_step"])
    same_p = all(np.array_equal(np.load(f"{W5}/{d_}nn_{names[3]}{s_}"), np.load(f"{W5}/{d_}nn_{names[3]}@last{s_}"))
                 for d_, s_ in (("preds/", "_trainonly_val.npy"), ("preds/", "_trainonly_test.npy"),
                                ("preds/holdout/", "_holdout.npy")))
    check("arms ON: @last val / test / holdout preds == the selected checkpoint's (same weights)", same_p)
    check("arms ON: setup log lines (TRUNC-PROD, AUXF multiplier, CLIPLOG, selection, LAST checkpoint)",
          all(s in lc for s in ("wave-5 TRUNC-PROD: a 1.0", "wave-5 AUXF:", "wave-5 CLIPLOG final",
                                "wave-5 selection: checkpoint chosen only among evals at step >= 39",
                                "wave-5 LAST checkpoint (step 39)", "wave-5 saved last-checkpoint test preds")),
          [ln for ln in lc.splitlines() if "wave-5" in ln][:3])
    check("arms ON changes training (checkpoint differs from flags OFF)",
          not ck_equal(f"{W5}/ckpt/{names[3]}.pt", f"{W5}/ckpt/{names[1]}.pt"))
    RES["e2e"] = dict(off_lines=mb, on_lines=metric_lines(lc), w5_lines=[ln for ln in lc.splitlines() if "wave-5" in ln])
    for v in names:
        clean(v)


def main():
    t_all = time.time()
    P(f"== nn_x/w5 wave-5 tests, part {PART} ({time.ctime()}, device {DEV}, CPU threads {NT})")
    meta = train_meta()
    if PART == "cpu":
        c0_code(load_parse(TRAIN_PY))
        c3_auxf(meta)
        c2_trunc(meta)
        c1_artifacts(meta)
    else:
        g1_auxf()
        g2_e2e(meta)
    RES["failed"] = FAIL
    P(f"\n== {len(RES['checks']) - len(FAIL)}/{len(RES['checks'])} checks passed; failed: {FAIL}; "
      f"{time.time() - t_all:.0f}s")
    json.dump(RES, open(f"{W5}/logs/test_w5_{PART}.json", "w"), indent=1, default=str)
    return 0 if not FAIL else 1
